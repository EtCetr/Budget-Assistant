class AuthNotifier extends _$AuthNotifier {
  late final IAuthRepository _repository;
  StreamSubscription<AuthState>? _authStreamSubscription;

  @override
  AuthStatus build() {
    _repository = ref.watch(authRepositoryProvider);
    ref.onDispose(() {
      _authStreamSubscription?.cancel();
    });
    _initAuthState();
    return AuthStatus.unknown;
  }

  void _initAuthState() {
    _authStreamSubscription = _repository.authStateChanges.listen(
      _handleAuthStateChange,
    );
  }

  void _handleAuthStateChange(AuthState authState) {
    if (authState.session != null) {
      state = AuthStatus.authenticated;
    } else {
      state = AuthStatus.unauthenticated;
    }
  }

  Future<Result<void>> signInWithEmailAndPassword(
    String email,

  ) async {
    return _repository.signUpWithEmailAndPassword(email, password);
  }

@DriftAccessor(tables: [YearlyAggregates])
class YearlyAggregatesDao extends DatabaseAccessor<AppDatabase>
    with _$YearlyAggregatesDaoMixin {
  YearlyAggregatesDao(super.db);

  Future<YearlyAggregateDb?> getForScope({
    required String userId,
    required String? spaceId,
    required int year,
  }) {
    final query = select(yearlyAggregates)
      ..where((t) => t.userId.equals(userId) & t.year.equals(year));
    if (spaceId == null) {
      query.where((t) => t.spaceId.isNull());
    } else {
      query.where((t) => t.spaceId.equals(spaceId));
    }
    return query.getSingleOrNull();
  }

  Future<void> upsertAggregate({
    required String newId,
    required String userId,
    required String? spaceId,
    required int year,
    required String dataJson,
  }) async {
    final existing =
        await getForScope(userId: userId, spaceId: spaceId, year: year);
    final now = DateTime.now().toUtc();
    if (existing == null) {
      await into(yearlyAggregates).insert(
        YearlyAggregatesCompanion.insert(
          id: newId,
          userId: userId,
          year: year,
          dataJson: dataJson,
          lastCalculatedAt: now,
          spaceId: Value(spaceId),
        ),
      );
      return;
    }
    await (update(yearlyAggregates)
          ..where((t) => t.id.equals(existing.id)))
        .write(
      YearlyAggregatesCompanion(
        dataJson: Value(dataJson),

class TransferProfileNotifier extends Notifier<TransferProfile> {
  static const _key = 'transfer_profile_json';

  @override
  TransferProfile build() => const TransferProfile();

  Future<TransferProfile> load() async {
    try {
      final raw = await SecureStorageService().read(_key);
      if (raw != null && raw.isNotEmpty) {
        state = TransferProfile.fromJson(
            (jsonDecode(raw) as Map).cast<String, dynamic>());
      }
    } catch (_) {}
    return state;
  }

  Future<void> save(TransferProfile next) async {
    state = next;
    try {
      await SecureStorageService().write(_key, jsonEncode(next.toJson()));
    } catch (_) {}
  }
}

    NotifierProvider<TransferProfileNotifier, TransferProfile>(
        TransferProfileNotifier.new);
