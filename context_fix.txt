enum SavingsExportFormat { xlsx, csv, png, pdf }

class ExportSavingsAnalyticsUseCase {
  ExportSavingsAnalyticsUseCase({required Logger logger}) : _logger = logger;
  final Logger _logger;

  Future<String> call({
    required SavingsExportFormat format,
    required List<SavingsGoal> goals,
    required String baseCurrency,
    GlobalKey? chartKey,
  }) async {
    try {
      final dir = await getTemporaryDirectory();
      final stamp = DateTime.now().toUtc().toIso8601String().substring(0, 10);
      switch (format) {
        case SavingsExportFormat.xlsx:
          return _xlsx(dir, stamp, goals, baseCurrency);
        case SavingsExportFormat.csv:
          return _csv(dir, stamp, goals, baseCurrency);
        case SavingsExportFormat.pdf:
          return _pdf(dir, stamp, goals, baseCurrency);
        case SavingsExportFormat.png:
          return _png(dir, stamp, chartKey);
      }
    } catch (e, st) {
      _logger.e('ExportSavingsAnalyticsUseCase failed', error: e, stackTrace: st);
      rethrow;
    }
  }

  Future<String> _xlsx(Directory dir, String stamp, List<SavingsGoal> goals, String cur) async {
    final excel = Excel.createExcel();
    final sheetGoals = excel['Р¦РµР»Рё'];
    sheetGoals.appendRow([TextCellValue('Р¦РµР»СЊ'), TextCellValue('Р¦РµР»РµРІР°СЏ'), TextCellValue('РќР°РєРѕРїР»РµРЅРѕ'), TextCellValue('%'), TextCellValue('Р”РµРґР»Р°Р№РЅ'), TextCellValue('РЎС‚Р°С‚СѓСЃ')]);
    for (final g in goals) {
      sheetGoals.appendRow([
        TextCellValue(g.name),
        TextCellValue((g.targetAmount / 100).toStringAsFixed(2)),
        TextCellValue((g.currentAmount / 100).toStringAsFixed(2)),
        TextCellValue('${g.progressPercent}'),
        TextCellValue(g.deadline?.toIso8601String() ?? ''),
        TextCellValue(g.status),
      ]);
    }
    final path = '${dir.path}/savings_analytics_$stamp.xlsx';
    excel.save(path);
    return path;
  }

  Future<String> _csv(Directory dir, String stamp, List<SavingsGoal> goals, String cur) async {
    final sb = StringBuffer();
    sb.writeln('Р¦РµР»СЊ,Р¦РµР»РµРІР°СЏ,РќР°РєРѕРїР»РµРЅРѕ,%,Р”РµРґР»Р°Р№РЅ,РЎС‚Р°С‚СѓСЃ');
    for (final g in goals) {
      sb.writeln([
        '"${g.name.replaceAll('"', '""')}"',
        (g.targetAmount / 100).toStringAsFixed(2),
        (g.currentAmount / 100).toStringAsFixed(2),
        '${g.progressPercent}',
        g.deadline?.toIso8601String() ?? '',
        g.status,
      ].join(','));
    }
    final path = '${dir.path}/savings_goals_$stamp.csv';
    // UTF-8 СЃ BOM РґР»СЏ Excel
    await File(path).writeAsBytes([0xEF, 0xBB, 0xBF] + sb.toString().codeUnits);
    return path;
  }

  Future<String> _pdf(Directory dir, String stamp, List<SavingsGoal> goals, String cur) async {
    final pdf = pw.Document();
    pdf.addPage(pw.TableHelper.fromTextArray(
      header: ['Р¦РµР»СЊ', 'Р¦РµР»РµРІР°СЏ', 'РќР°РєРѕРїР»РµРЅРѕ', '%', 'РЎС‚Р°С‚СѓСЃ'],
      data: goals.map((g) => [
        g.name,
        (g.targetAmount / 100).toStringAsFixed(2),
        (g.currentAmount / 100).toStringAsFixed(2),
        '${g.progressPercent}',
        g.status,
      ]).toList(),
    ));
    final path = '${dir.path}/savings_analytics_$stamp.pdf';
    await File(path).writeAsBytes(await pdf.save());
    return path;
  }

  Future<String> _png(Directory dir, String stamp, GlobalKey? chartKey) async {
    if (chartKey == null) throw StateError('chartKey is null');
    final boundary = chartKey.currentContext?.findRenderObject() as RepaintBoundary?;
    if (boundary == null) throw StateError('RepaintBoundary not found');
    final image = await boundary.toImage(pixelRatio: 3.0);
    final bytes = await image.toByteData(format: ui.ImageByteFormat.png);
    if (bytes == null) throw StateError('PNG encode failed');
    final path = '${dir.path}/savings_chart_$stamp.png';
    await File(path).writeAsBytes(bytes.buffer.asUint8List());
    return path;
  }

  Future<void> share(String path) async {
    await SharePlus.instance.share(ShareParams(files: [XFile(path)], text: 'РњРѕСЏ Р°РЅР°Р»РёС‚РёРєР° РєРѕРїРёР»РѕРє СЃ Budget Assistant рџљЂ'));
  }
}

class AnalyticsPeriodRange {
  const AnalyticsPeriodRange({
    required this.start,
    required this.end,
    required this.prevStart,
    required this.prevEnd,
  });

  final DateTime start;
  final DateTime end;
  final DateTime prevStart;
  final DateTime prevEnd;

  int get totalDays => end.difference(start).inDays + 1;
}

final analyticsPeriodRangeProvider = Provider<AnalyticsPeriodRange>((ref) {
  final period = ref.watch(analyticsPeriodProvider);
  final now = DateTime.now();
  final DateTime start;
  switch (period) {
    case AnalyticsPeriod.month:
      start = DateTime(now.year, now.month, 1);
    case AnalyticsPeriod.quarter:
      start = DateTime(now.year, ((now.month - 1) ~/ 3) * 3 + 1, 1);
    case AnalyticsPeriod.year:
      start = DateTime(now.year, 1, 1);
    case AnalyticsPeriod.allTime:
      start = DateTime(2000, 1, 1);
  }
  final DateTime prevStart;
  final DateTime prevEnd;
  if (period == AnalyticsPeriod.allTime) {
    prevStart = start;
    prevEnd = start.subtract(const Duration(days: 1));
  } else {
    final days = now.difference(start).inDays + 1;
    prevEnd = start.subtract(const Duration(days: 1));
    prevStart = prevEnd.subtract(Duration(days: days - 1));
  }
  return AnalyticsPeriodRange(
    start: start,
    end: now,
    prevStart: prevStart,
    prevEnd: prevEnd,
  );
});
